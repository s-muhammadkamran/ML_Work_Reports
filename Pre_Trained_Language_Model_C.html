In this assignment, you will work on the ComVE shared task that was part of SemEval-2020. The task aims to evaluate whether a system can distinguish if a natural language statement makes sense to humans or not and provide a reason. ConVE includes three subtasks that require models to acquire and apply commonsense knowledge. In this notebook you will focus on SubTask C:

  • Given a statement that does not make sense, generate the reason why this statement does not make sense. For each nonsensical statement, three valid reasons are given as reference:

    Statement: He put an elephant into the fridge.
    Reason A: An elephant is much bigger than a fridge.
    Reason B: A fridge is much smaller than an elephant.
    Reason C: Most of the fridges aren't large enough to contain an elephant.

    This subtask can be approached as a Sequence-to-Sequence problem where the input is the nonsensical statement and the output is a valid reason.

def preprocess_data(examples, tokenizer, max_length, is_test=False):   # [1 Mark]

    if not is_test:

            labels = tokenizer(

                padding="max_length",

    return model_inputs

def evaluate_prediction(test_data, metric):   # [2 Marks]
    #
    #  REPLACE THE pass STATEMENT WITH YOUR CODE
    #
    metric_name = str(metric).lower()
    preds = test_data["prediction"].astype(str).tolist()

    if metric_name == "bleu":
        bleu = evaluate.load("bleu")
        # references: list of list of strings
        refs = test_data[["reason1", "reason2", "reason3"]].fillna("").values.tolist()
        # Remove empty strings within each reference set
        refs = [[r for r in ref if str(r).strip() != ""] for ref in refs]
        result = bleu.compute(predictions=preds, references=refs)
        return {"bleu": result.get("bleu", None)}

    elif metric_name == "rouge":
        rouge = evaluate.load("rouge")
        # Compute per-example max rougeLsum across the references
        best_scores = []
        for _, row in test_data.iterrows():
            cand = str(row["prediction"])
            refs = [str(row.get("reason1", "")), str(row.get("reason2", "")), str(row.get("reason3", ""))]
            refs = [r for r in refs if r.strip() != ""]
            if not refs:
                best_scores.append(0.0)
                continue
            scores = []
            for r in refs:
                out = rouge.compute(predictions=[cand], references=[r])
                # Prefer rougeLsum if available; fallback to rougeL
                score = out.get("rougeLsum", None)
                if score is None:
                    score = out.get("rougeL", 0.0)
                scores.append(score)
            best_scores.append(max(scores))
        avg_rouge = sum(best_scores) / len(best_scores) if best_scores else 0.0
        return {"rougeL(best_ref_avg)": avg_rouge}

    else:
        raise ValueError("Unsupported metric: use 'bleu' or 'rouge'")
